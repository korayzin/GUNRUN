Ekran görüntünde gördüm - LeaderboardRow var ama panel yok.
O zaman LeaderboardRow = Ana Panel olacak!

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⚡ ADIM 1: LeaderboardRow'a Script Ekle (30 saniye)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Hierarchy'de "LeaderboardRow" seç
2. Inspector → Add Component → "LeaderboardUI" yaz ve ekle
3. Bitti! ✅

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⚡ ADIM 2: Referansları Ata (1 dakika)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

LeaderboardUI component'inde sadece 2 şey ata:

   ┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
   ┃  1. Leaderboard Content  →  Scroll View/Viewport/Content  ┃ ⭐
   ┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛
        
        Nasıl yapılır:
        - Hierarchy'de LeaderboardRow'u aç (ok işaretine tıkla)
        - Score'u aç
        - Scroll View'ı aç
        - Viewport'u aç
        - "Content" objesini bul
        - Content'i LeaderboardUI → Leaderboard Content alanına sürükle
   
   ┏━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┓
   ┃  2. Leaderboard Panel  →  LeaderboardRow (kendisi)        ┃
   ┗━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━┛
        
        Nasıl yapılır:
        - LeaderboardRow'u LeaderboardUI → Leaderboard Panel'e sürükle

Diğer alanlar boş kalabilir (otomatik bulunacak veya opsiyonel)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

⚡ ADIM 3: Firebase Manager Ekle (30 saniye)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Hierarchy → Sağ tık → Create Empty
2. İsim: "FirebaseManager"
3. Add Component → "FirebaseLeaderboardManager"
4. Bitti! URL zaten doğru ✅

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

✅ TAMAMDIR! Şimdi test et:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Play mode → Console aç → Bu kodu çalıştır:

FindObjectOfType<LeaderboardUI>().AddTestPlayers();

2 saniye bekle, sonra:

FindObjectOfType<LeaderboardUI>().LoadLeaderboard();

Scroll View içinde oyuncular görünecek! 🎉
